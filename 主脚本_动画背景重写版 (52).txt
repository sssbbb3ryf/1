        local hiddenFolder = game:FindFirstChild("__HiddenObjects__")
        if not hiddenFolder then
            hiddenFolder = Instance.new("Folder")
            hiddenFolder.Name = "__HiddenObjects__"
            hiddenFolder.Parent = game
        end

        local targetPaths = {
            "Things.BomberBarry",
            "Things.ProClimber",
            "Things.ToolWorker",
            "Things.GroupReward",
            "Things.StockProduct",
            "Things.Plots.Slots",
        }

        local blockedData = {}

        local function forceRefresh(instance)
            for _, child in ipairs(instance:GetChildren()) do
                forceRefresh(child)
            end
            local _ = instance.Name
            if instance:IsA("BasePart") then
                _ = instance.Position
            end
        end

        local function applyBlock()
            for _, path in ipairs(targetPaths) do
                local obj = workspace
                local segments = {}
                for segment in string.gmatch(path, "[^%.]+") do
                    table.insert(segments, segment)
                    obj = obj and obj:FindFirstChild(segment)
                    if not obj then break end
                end
                if obj then
                    if obj.Parent ~= hiddenFolder then
                        blockedData[obj] = { path = path, segments = segments }
                        obj.Parent = hiddenFolder
                    else
                        if not blockedData[obj] then
                            blockedData[obj] = { path = path, segments = segments }
                        end
                    end
                end
            end
        end

        local function removeBlock()
            local objectsToRefresh = {}
            for obj, data in pairs(blockedData) do
                if obj and obj.Parent == hiddenFolder then
                    local parent = workspace
                    local segments = data.segments
                    for i = 1, #segments - 1 do
                        local seg = segments[i]
                        local child = parent:FindFirstChild(seg)
                        if child then
                            parent = child
                        else
                            parent = workspace
                            break
                        end
                    end
                    obj.Parent = parent
                    table.insert(objectsToRefresh, obj)
                end
            end
            blockedData = {}
            for _, obj in ipairs(objectsToRefresh) do
                forceRefresh(obj)
            end
            task.wait(0.5)
        end

        local posStack = {}
        local function pushPosition(cf) table.insert(posStack, cf) end
        local function popPosition() return table.remove(posStack) end

        -- 自动出售：检测到 Sell 弹窗后自动执行，传送→持续出售1秒→自动传回
        local sellTargetCF = CFrame.new(
            -46.0041466, 28.2632484, 1066.44812,
            -0.00915288925, -0.0258089378, 0.999624968,
            -0.0115938438, 0.999602437, 0.0257021971,
            -0.999890924, -0.0113542452, -0.00944840908
        )
        local sellDuration = 1  -- 出售持续时间（秒）
        local sellInterval = 0.1  -- 出售指令间隔（秒）

        local sellPaused = false  -- 矿石传送期间暂停自动出售
        local sellEnabled = false  -- 自动出售开关，默认关闭

        local function autoSell()
            if selling or sellPaused then return end
            if (tick() - lastSellTime) < 5 then return end
            local char = player.Character
            if not char then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then return end

            selling = true
            local savedPos = hrp.CFrame
            lastSellTime = tick()
            print("[出售] 检测到出售界面，自动执行出售...")

            -- 传送到出售点
            hrp.CFrame = sellTargetCF
            task.wait(0.1)

            -- 持续发送出售指令，持续 sellDuration 秒
            local startTime = tick()
            local sellCount = 0
            while (tick() - startTime) < sellDuration do
                pcall(function()
                    local remote = ReplicatedStorage:FindFirstChild("Remotes")
                    if remote then
                        local sr = remote:FindFirstChild("SellRequest")
                        if sr then sr:FireServer("all") end
                    end
                end)
                sellCount = sellCount + 1
                task.wait(sellInterval)
            end

            -- 传送回原位（等角色稳定，最多重试10次）
            for retry = 1, 10 do
                char = player.Character
                if char then
                    hrp = char:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        hrp.CFrame = savedPos
                        break
                    end
                end
                task.wait(0.3)
            end

            -- 强制关闭出售 UI
            pcall(function()
                local sell = playerGui:FindFirstChild("Sell")
                if sell then
                    local frame = sell:FindFirstChild("Frame")
                    if frame and frame.Visible then
                        frame.Visible = false
                    end
                end
            end)
            -- 同步 hook 状态为已关闭
            sellWasShowing = false

            selling = false
            print("[出售] 出售完成（发送" .. sellCount .. "次指令），已自动返回原位")
        end

        -- 监听 PlayerGui.Sell.Frame 的 Visible 变化
        local sellWasShowing = false

        local function hookSellFrame(sellFrame)
            if not sellFrame then return end
            sellWasShowing = sellFrame.Visible
            pcall(function()
                sellFrame:GetPropertyChangedSignal("Visible"):Connect(function()
                    if selling or not sellEnabled then return end
                    if sellFrame.Visible and not sellWasShowing then
                        sellWasShowing = true
                        task.spawn(function()
                            task.wait(0.15)
                            autoSell()
                        end)
                    elseif not sellFrame.Visible then
                        sellWasShowing = false
                    end
                end)
            end)
        end

        local function tryHookSell()
            local sell = playerGui:FindFirstChild("Sell")
            if sell then
                local frame = sell:FindFirstChild("Frame")
                if frame then
                    hookSellFrame(frame)
                    return true
                end
            end
            return false
        end

        -- 如果 Sell.Frame 还不存在，等它出现
        if not tryHookSell() then
            -- 先等 Sell 出现
            local sellObj = playerGui:FindFirstChild("Sell")
            if not sellObj then
                playerGui.ChildAdded:Connect(function(child)
                    if child.Name == "Sell" then
                        local frame = child:WaitForChild("Frame", 5)
                        if frame then
                            hookSellFrame(frame)
                        end
                    end
                end)
            else
                -- Sell 存在但 Frame 还没加载
                sellObj.ChildAdded:Connect(function(child)
                    if child.Name == "Frame" then
                        hookSellFrame(child)
                    end
                end)
            end
        end

        -- 自动收集
        local collectActive = false
        local collectThread = nil
        local rangePart = nil
        local heartbeatConnection = nil

        local function createRangePart()
            if rangePart then return end
            rangePart = Instance.new("Part")
            rangePart.Size = Vector3.new(36, 36, 36)
            rangePart.Shape = Enum.PartType.Ball
            rangePart.Transparency = 1
            rangePart.Anchored = true
            rangePart.CanCollide = false
            rangePart.Material = Enum.Material.SmoothPlastic
            rangePart.BrickColor = BrickColor.new("Bright blue")
            rangePart.Parent = workspace
        end

        local function destroyRangePart()
            if rangePart then
                rangePart:Destroy()
                rangePart = nil
            end
        end

        local function getPromptPart(prompt)
            local part = prompt:FindFirstAncestorOfClass("BasePart")
            if part then return part end
            if prompt.Parent and prompt.Parent:IsA("BasePart") then
                return prompt.Parent
            end
            local parent = prompt.Parent
            if parent then
                for _, child in ipairs(parent:GetDescendants()) do
                    if child:IsA("BasePart") then
                        return child
                    end
                end
            end
            return nil
        end

        local function isInRange(part)
            if not rangePart or not part then return false end
            local center = rangePart.Position
            local radius = 18
            local distance = (part.Position - center).Magnitude
            return distance <= radius
        end

        local function startCollect()
            if collectThread then return end
            collectActive = true
            applyBlock()
            task.wait(0.5)
            createRangePart()
            task.wait(0.5)

            heartbeatConnection = RunService.Heartbeat:Connect(function()
                if not collectActive then return end
                local char = player.Character
                if char and rangePart then
                    local hrp = char:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        rangePart.CFrame = hrp.CFrame
                    end
                end
            end)

            collectThread = coroutine.create(function()
                while collectActive do
                    local char = player.Character
                    if char and rangePart then
                        local hrp = char:FindFirstChild("HumanoidRootPart")
                        if hrp then
                            local runtime = workspace:FindFirstChild("Runtime")
                            local scanRoot = runtime and runtime:FindFirstChild("LootPoints") or workspace
                            for _, obj in ipairs(scanRoot:GetDescendants()) do
                                if obj:IsA("ProximityPrompt") then
                                    local part = getPromptPart(obj)
                                    if part and isInRange(part) then
                                        pcall(function() fireproximityprompt(obj) end)
                                    end
                                end
                            end

        local function parseObjectText(text)
            if not text or text == "" then return nil end
            local parts = {}
            for part in string.gmatch(text, "[^•]+") do
                local trimmed = part:gsub("^%s+", ""):gsub("%s+$", "")
                table.insert(parts, trimmed)
            end
            if #parts >= 3 then
                local namePart = parts[1]
                local name = namePart:match("%S+$") or namePart
                return {
                    name = name,
                    weight = parts[2],
                    price = parts[3]
                }
            end
            return nil
        end

        local function getQualityFromName(name)
            if not name then return 0 end
            local lower = name:lower()
            if lower:find("crystal_t1") then return 0 end
            if lower:find("crystal_t2") then return 1 end
            if lower:find("crystal_t3") then return 2 end
            if lower:find("crystal_t4") then return 3 end
            if lower:find("crystal_t5") then return 4 end
            if lower:find("crystal_t6") then return 5 end
            return 0
        end

                                    oldData.card.frame.BackgroundTransparency = 1
                                end
                            end
                            selectedOre = part
                            if data and data.card then
                                data.card.frame.BackgroundTransparency = 0.65
                            end
                            teleportBtn.TextColor3 = Color3.fromRGB(255, 182, 193)
                            returnBtn.TextColor3 = Color3.new(0.5, 0.5, 0.5)
                        end
                        tpLastClickTime = now
                    end)

                    data = {
                        billboard = billboard,
                        anchor = anchor,
                        card = card,
                        clickDetector = clickDetector,
                        tpRow = tpRow,
                        tpName = tpName,
                        tpBtn = tpBtn,
                    }
                    oreData[part] = data
                end

                local card = data.card
                card.nameLabel.Text = displayName
                card.distLabel.Text = string.format("%.1f m", distance)
                card.priceLabel.Text = price
                local qualityName = QUALITY_NAMES[qualityLevel + 1] or "普通"

                else

                    data.tpName.Text = displayName .. " [" .. qualityName .. "]"
                    data.tpName.TextColor3 = currentColor
                    
                    data.sortQuality = qualityLevel

            end
            
            table.sort(visibleOres, function(a, b)

                    return a.sortQuality > b.sortQuality
                else
                    return a.sortWeight > b.sortWeight
                end

            for index, oreDataEntry in ipairs(visibleOres) do

                end
            end
        end

-- ============================================================
-- 加载动画：自动检测帧数 + BGM（只播放一次，最顶层遮挡一切）
-- 统一走本地缓存（PC和手机端都需要先缓存到本地才能播放）
-- 手机端无文件系统时提示用户先运行预下载脚本
-- ============================================================
local function playLoadingAnimation()
    -- GitHub raw 链接（无302重定向，Roblox可直接下载）
    local LOADING_BASE = "https://raw.githubusercontent.com/sssbbb3ryf/loading/main/"
    local LOADING_CACHE = "loading_anim_cache"
    local BGM_FILE = "loading_bgm.mp3"
    if makefolder then makefolder(LOADING_CACHE) end

    -- 检测是否支持文件系统（PC和部分手机执行器支持，不支持则提示先运行预下载脚本）
    local hasFileSystem = (isfile and readfile and writefile and getcustomasset)

    -- 检测本地缓存是否完整（至少有1帧才算可用）
    local function checkLocalCache()
        if not hasFileSystem then return 0 end
        local count = 0
        for i = 1, 1000 do
            local fname = string.format("loading_%04d.jpg", i)
            if isfile(LOADING_CACHE .. "/" .. fname) then
                count = i
            else
                break
            end
        end
        return count
    end

    local skipPressed = false
    local bgmSound = nil

    -- 带超时的 HttpGet（防止卡死）
    local function httpGetSafe(url, timeoutSec)
        timeoutSec = timeoutSec or 10
        local result = nil
        local done = false
        task.spawn(function()
            local ok, data = pcall(function()
                return game:HttpGet(url)
            end)
            if ok then result = data end
            done = true
        end)
        local start = tick()
        while not done do
            task.wait(0.1)
            if (tick() - start) > timeoutSec then
                done = true
                break
            end
        end
        return result
    end

    -- 自动检测可用帧数：优先本地缓存，其次网络探测
    local function detectFrameCount()
        -- 优先检查本地缓存
        local cached = checkLocalCache()
        if cached > 0 then
            print("[加载] 本地缓存已有 " .. cached .. " 帧，直接使用")
            return cached
        end
        -- 网络探测：从1开始，每50帧跳着检测
        local lastFound = 0
        for i = 1, 1000, 50 do
            local fname = string.format("loading_%04d.jpg", i)
            local data = httpGetSafe(LOADING_BASE .. fname, 5)
            if data and #data > 500 then
                lastFound = i
            else
                break
            end
        end
        if lastFound == 0 then return 0 end
        -- 精扫：从lastFound到lastFound+50逐个检测
        for i = lastFound + 1, lastFound + 50 do
            local fname = string.format("loading_%04d.jpg", i)
            local data = httpGetSafe(LOADING_BASE .. fname, 5)
            if data and #data > 500 then
                lastFound = i
            else
                break
            end
        end
        return lastFound
    end

    -- 后台静默准备
    task.spawn(function()
        print("[加载] 正在检测仓库帧数...")
        local LOADING_TOTAL = detectFrameCount()
        print("[加载] 检测到 " .. LOADING_TOTAL .. " 帧")

        if LOADING_TOTAL == 0 then
            print("[加载] 未检测到任何帧，跳过加载动画")
            createMainUI()
            return
        end

        -- ============================================================
        -- 帧准备：统一走本地缓存
        -- 无文件系统 → 提示运行预下载脚本，跳过动画
        -- 有文件系统但无缓存 → 提示运行预下载脚本，跳过动画
        -- 有文件系统且有缓存 → 直接用 getcustomasset 加载
        -- ============================================================
        local frameUrls = {}

        if not hasFileSystem then
            -- 无文件系统权限，无法播放动画
            print("[加载] 当前执行器不支持文件系统，无法播放加载动画")
            print("[加载] 请先运行「预下载脚本」来缓存动画资源，或更换支持文件系统的执行器")
            -- 弹出提示
            pcall(function()
                starterGui:SetCore("SendNotification", {
                    Title = "加载动画",
                    Text = "当前执行器不支持文件系统，请先运行预下载脚本！动画已跳过",
                    Duration = 10
                })
            end)
            createMainUI()
            return
        end

        -- 有文件系统，检查本地缓存
        local cachedCount = checkLocalCache()
        if cachedCount < LOADING_TOTAL then
            -- 缓存不完整或为空，提示运行预下载脚本
            print("[加载] 本地缓存不完整 (" .. cachedCount .. "/" .. LOADING_TOTAL .. ")")
            print("[加载] 请先运行「预下载脚本」来下载所有动画帧！动画已跳过")
            pcall(function()
                starterGui:SetCore("SendNotification", {
                    Title = "加载动画",
                    Text = "缓存不完整(" .. cachedCount .. "/" .. LOADING_TOTAL .. ")，请先运行预下载脚本！",
                    Duration = 10
                })
            end)
            createMainUI()
            return
        end

        -- 缓存完整，构建 getcustomasset URL 列表
        print("[加载] 本地缓存完整，直接使用 (" .. cachedCount .. " 帧)")
        for i = 1, LOADING_TOTAL do
            local fname = string.format("loading_%04d.jpg", i)
            local fpath = LOADING_CACHE .. "/" .. fname
            frameUrls[i] = getcustomasset(fpath)
        end

        local readyCount = #frameUrls
        print("[加载] 就绪: " .. readyCount .. "/" .. LOADING_TOTAL)

        if readyCount == 0 then
            print("[加载] 没有任何可用帧，跳过动画")
            createMainUI()
            return
        end

        -- ========================================================
        -- 全部准备完毕！创建UI
        -- ========================================================
        print("[加载] 全部就绪，创建UI并播放动画")

        starterGui:SetCoreGuiEnabled(Enum.CoreGuiType.All, false)
        pcall(function() starterGui:SetCore("TopbarEnabled", false) end)

        local screenGui = Instance.new("ScreenGui")
        screenGui.Name = "LoadingAnimation"
        screenGui.ResetOnSpawn = false
        screenGui.IgnoreGuiInset = true
        screenGui.DisplayOrder = 99999
        screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
        local ok, err = pcall(function()
            screenGui.Parent = game:GetService("CoreGui")
        end)
        if not ok then
            screenGui.Parent = playerGui
        end

        -- 黑色底板
        local bgFrame = Instance.new("Frame")
        bgFrame.Name = "BlackBG"
        bgFrame.Size = UDim2.new(1, 0, 1, 0)
        bgFrame.Position = UDim2.new(0, 0, 0, 0)
        bgFrame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
        bgFrame.BackgroundTransparency = 0
        bgFrame.ZIndex = 0
        bgFrame.Parent = screenGui

        -- 创建所有帧的 ImageLabel（PC和手机端统一方案）
        -- 先创建并赋值URL，然后 PreloadAsync 强制下载到内存
        local frames = {}
        for i = 1, readyCount do
            local lbl = Instance.new("ImageLabel")
            lbl.Name = "LoadF" .. i
            lbl.Size = UDim2.new(1, 0, 1, 0)
            lbl.Position = UDim2.new(0, 0, 0, 0)
            lbl.BackgroundTransparency = 1
            lbl.ImageTransparency = 1
            lbl.ZIndex = 5
            lbl.ScaleType = Enum.ScaleType.Stretch
            if frameUrls[i] then
                lbl.Image = frameUrls[i]
            end
            lbl.Parent = screenGui
            frames[i] = lbl
        end

        -- 显示第一帧
        if frames[1] then
            frames[1].ImageTransparency = 0
        end

        -- GPU预加载：强制下载所有URL图片到GPU内存
        print("[加载] GPU 预加载 " .. #frames .. " 帧...")
        pcall(function()
            local ContentProvider = game:GetService("ContentProvider")
            ContentProvider:PreloadAsync(frames)
        end)
        print("[加载] GPU 预加载完成，开始播放动画")

        -- 跳过按钮
        local skipBtn = Instance.new("TextButton")
        skipBtn.Size = UDim2.new(0, 100, 0, 40)
        skipBtn.Position = UDim2.new(1, -120, 1, -60)
        skipBtn.AnchorPoint = Vector2.new(1, 1)
        skipBtn.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
        skipBtn.BackgroundTransparency = 0.5
        skipBtn.Text = "跳过"
        skipBtn.TextColor3 = Color3.new(1, 1, 1)
        skipBtn.TextSize = 20
        skipBtn.Font = Enum.Font.SourceSansBold
        skipBtn.ZIndex = 100
        skipBtn.Parent = screenGui

        -- 停止 BGM 并清理
        local function finishLoading()
            if skipPressed then return end
            skipPressed = true
            if bgmSound then
                bgmSound:Stop()
                bgmSound:Destroy()
                bgmSound = nil
            end
            starterGui:SetCoreGuiEnabled(Enum.CoreGuiType.All, true)
            pcall(function() starterGui:SetCore("TopbarEnabled", true) end)
            if screenGui then
                screenGui:Destroy()
            end
            print("加载动画完毕，开始创建主 UI")
            createMainUI()
        end

        skipBtn.MouseButton1Click:Connect(function()
            if skipPressed then return end
            finishLoading()
        end)

        -- 播放BGM（本地缓存）
        pcall(function()
            bgmSound = Instance.new("Sound")
            if isfile(BGM_FILE) then
                bgmSound.SoundId = getcustomasset(BGM_FILE)
            else
                bgmSound.SoundId = LOADING_BASE .. "loading_bgm.mp3"
            end
            bgmSound.Volume = 0.5
            bgmSound.Looped = false
            bgmSound.Parent = workspace
            bgmSound:Play()
            print("[加载] BGM 已开始播放")
        end)

        print("[加载] 开始播放动画（一次）...")

        -- 播放：切换透明度，30fps，播放一次后淡出
        local cur = 1
        local lastSwitch = 0
        local FPS = 30
        local interval = 1 / FPS
        local finished = false

        local animConn
        animConn = RunService.RenderStepped:Connect(function()
            if skipPressed or not screenGui or not screenGui.Parent then
                if animConn then animConn:Disconnect() end
                return
            end
            if finished then return end
            local now = tick()
            if now - lastSwitch >= interval then
                lastSwitch = now
                if frames[cur] then
                    frames[cur].ImageTransparency = 1
                end
                cur = cur + 1
                if cur > readyCount then
                    finished = true
                    if animConn then animConn:Disconnect() end
                    local fadeInfo = TweenInfo.new(1.5, Enum.EasingStyle.Linear)
                    for _, lbl in ipairs(frames) do
                        pcall(function()
                            TweenService:Create(lbl, fadeInfo, {ImageTransparency = 1}):Play()
                        end)
                    end
                    task.wait(1.5)
                    if not skipPressed then finishLoading() end
                    return
                end
                if frames[cur] then
                    frames[cur].ImageTransparency = 0
                end
            end
        end)
    end)
end
